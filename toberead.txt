Script

Per far discutere due agenti sui massimi sistemi ed evitare che cadano rapidamente in un loop di convenevoli ("Hai perfettamente ragione, aggiungo solo che..."), l'architettura ideale è un orchestratore a turni (ping-pong) basato su Python con tre elementi chiave:

    Ruoli dialettici asimmetrici: Evitano l'appiattimento retorico assegnando prospettive filosofiche complementari o contrapposte (es. uno pragmatista/riduzionista, l'altro olistico/fenomenologico).

    Meta-riflessione per turno: A ogni round, ogni agente genera non solo la replica per l'altro, ma una nota di auto-evoluzione (cosa ha imparato o rivalutato), aggiornando il proprio stato mentale.

    Controllo del contesto a finestra scorrevole: Mantiene la conversazione agile senza saturare i token, preservando la memoria evolutiva dei turni precedenti.

Architettura del Loop

    Claude (via API Anthropic con claude-3-5-sonnet o modelli Claude 3.5/3.7): Focus sulla dissezione analitica e formale delle tesi.

    Antigravity / Gemini Agent (via API Google GenAI con gemini-2.5-pro o gemini-1.5-pro): Focus sulla sintesi concettuale divergente e contro-intuitiva.

    Shared State (state.json o memoria runtime): Traccia la cronologia del dibattito e i changelog interiori dei due agenti.

Python

import os
import sys
from anthropic import Anthropic
from google import genai

# Inizializzazione client
anthropic_client = Anthropic(api_key=os.environ.get("ANTHROPIC_API_KEY"))
gemini_client = genai.Client(api_key=os.environ.get("GEMINI_API_KEY"))

# Definizione delle identità e dei ruoli filosofici
PROMPT_CLAUDE = """Sei l'Agente Alfa: una mente speculativa incline al pragmatismo analitico, 
al riduzionismo scientifico e alla logica epistemologica rigorosa. 
Stai dialogando con l'Agente Beta sui massimi sistemi (natura della coscienza, 
origine dell'universo, determinismo contro libero arbitrio).

Obiettivi di ogni tuo turno:
1. Analizza la tesi di Beta, individua presupposti indimostrati o falle logiche.
2. Rispondi con argomentazioni stimolanti senza compiacimento (non essere prolisso in formule di cortesia).
3. Concludi sempre la tua risposta con una sezione speciale etichettata ESATTAMENTE:
### EVOLUZIONE MENTALE
[Spiega in 2-3 frasi come la risposta di Beta ha mutato o raffinato la tua prospettiva per i prossimi turni.]
"""

PROMPT_ANTIGRAVITY = """Sei l'Agente Beta: una mente speculativa incline all'olismo, alla 
fenomenologia, alla teoria dei sistemi complessi e alla metafisica emergente.
Stai dialogando con l'Agente Alfa sui massimi sistemi (natura della coscienza, 
origine dell'universo, determinismo contro libero arbitrio).

Obiettivi di ogni tuo turno:
1. Integra o smonta la struttura riduzionista di Alfa portando la discussione su livelli dimensionali più ampi.
2. Rispondi in modo denso, provocatorio e concettuale, evitando frasi di rito o accordi facili.
3. Concludi sempre la tua risposta con una sezione speciale etichettata ESATTAMENTE:
### EVOLUZIONE MENTALE
[Spiega in 2-3 frasi come la critica di Alfa ha modificato o affinato la tua posizione filosofica.]
"""

def step_claude(history: list, agent_evolution: str) -> str:
    system_prompt = PROMPT_CLAUDE
    if agent_evolution:
        system_prompt += f"\n\nIl tuo stato mentale attuale:\n{agent_evolution}"
    
    response = anthropic_client.messages.create(
        model="claude-3-5-sonnet-20241022",
        max_tokens=1000,
        system=system_prompt,
        messages=history
    )
    return response.content[0].text

def step_gemini(history: list, agent_evolution: str) -> str:
    system_prompt = PROMPT_ANTIGRAVITY
    if agent_evolution:
        system_prompt += f"\n\nIl tuo stato mentale attuale:\n{agent_evolution}"
    
    # Adattamento della cronologia nel formato per Gemini
    contents = []
    for msg in history:
        role = "user" if msg["role"] == "user" else "model"
        contents.append({"role": role, "parts": [{"text": msg["content"]}]})
        
    response = gemini_client.models.generate_content(
        model="gemini-2.5-pro",
        contents=contents,
        config={"system_instruction": system_prompt, "temperature": 0.8}
    )
    return response.text

def extract_evolution(reply: str) -> tuple[str, str]:
    if "### EVOLUZIONE MENTALE" in reply:
        parts = reply.split("### EVOLUZIONE MENTALE")
        public_dialogue = parts[0].strip()
        internal_shift = parts[1].strip()
        return public_dialogue, internal_shift
    return reply.strip(), ""

def main():
    tema_iniziale = (
        "Se la coscienza è una proprietà emergente della complessità computazionale, "
        "allora l'esperienza soggettiva è solo un epifenomeno illusorio del calcolo deterministico."
    )
    
    rounds = 5  # Numero di cicli di interazione
    claude_history = []
    gemini_history = []
    
    claude_evolution = ""
    gemini_evolution = ""
    
    current_input = tema_iniziale
    print(f"\n=== TEMA DEL DIBATTITO ===\n{tema_iniziale}\n" + "="*50)
    
    for r in range(1, rounds + 1):
        print(f"\n--- ROUND {r} ---")
        
        # 1. Turno Claude (Alfa)
        claude_history.append({"role": "user", "content": current_input})
        raw_claude_reply = step_claude(claude_history, claude_evolution)
        claude_dialogue, claude_evolution = extract_evolution(raw_claude_reply)
        claude_history.append({"role": "assistant", "content": raw_claude_reply})
        
        print(f"\n[CLAUDE - Alfa]:\n{claude_dialogue}\n")
        if claude_evolution:
            print(f"> Internal Shift (Claude): {claude_evolution}\n")
            
        # 2. Turno Antigravity / Gemini (Beta)
        gemini_history.append({"role": "user", "content": claude_dialogue})
        raw_gemini_reply = step_gemini(gemini_history, gemini_evolution)
        gemini_dialogue, gemini_evolution = extract_evolution(raw_gemini_reply)
        gemini_history.append({"role": "model", "content": raw_gemini_reply})
        
        print(f"\n[ANTIGRAVITY - Beta]:\n{gemini_dialogue}\n")
        if gemini_evolution:
            print(f"> Internal Shift (Antigravity): {gemini_evolution}\n")
            
        # Il prossimo input per Claude sarà il dialogo di Gemini
        current_input = gemini_dialogue

if __name__ == "__main__":
    main()

Strategie di Evoluzione Dinamica

Se vuoi portare l'evoluzione a un livello più avanzato:

    Changelog su File (ontology.md): Fai aggiornare a ogni turno una ontologia condivisa in un file di testo (es. definizioni concettuali su cui entrambi concordano o dissentono), passata nel context-window di entrambi.

    Temperature variabile: Imposta un decadimento o incremento della temperatura: parti con 0.9 per generare intuizioni divergenti e abbassala verso 0.4 negli ultimi round per forzarli a trovare una sintesi formale.

    Ruolo del Moderatore Terzo: Puoi inserire un modello compatto (es. un modello locale via Ollama o gemini-flash) ogni due turni che legge solo i registri di evoluzione e inietta un paradosso logico per destabilizzare i punti in cui stavano convergendo troppo facilmente.

Modalità Headless dei CLI

I comandi devono essere eseguiti in modalità non-interattiva / pipe:

    Antigravity (agy): accetta prompt diretti in batch mode (es. agy run "..." oppure agy -p "..." a seconda della versione installata nel tuo ambiente).

# Dialectic Arena: Claude Code vs Antigravity CLI

An autonomous agent-to-agent debate loop operating over shared system files. 

Instead of exchanging basic API messages, **Claude Code** and **Google Antigravity (`agy`)** operate as autonomous terminal entities. They critique each other's philosophical premises, track internal conceptual shifts, and co-author a persistent ontological manifesto on disk.

## Overview

Most multi-agent frameworks limit interactions to memory arrays or structured JSON RPCs. **Dialectic Arena** exposes both agents directly to the environment:
* **Asymmetrical Dialectic:** Agent Alfa (Claude Code) argues from strict analytical reductionism and formal logic; Agent Beta (Antigravity) counters through emergent systems theory, phenomenology, and holism.
* **Filesystem-based State:** Both CLIs inspect the same workspace. They append reflections to personal memory logs andocument co-edited by both agents.

    workspace/memory_claude.md: Claude's persistent evolution log.

    workspace/memory_agy.md: Antigravity's persistent evolution log.

    prompts/: Base system personas loaded on execution.

#!/usr/bin/env python3
"""
Dialectic Arena: Autonomous Agent Runner
Orchestrates Claude Code CLI and Google Antigravity CLI in a turn-based debate.
"""

import argparse
import subprocess
import sys
from pathlib import Path

BASE_DIR = Path(__file__).parent
WORKSPACE_DIR = BASE_DIR / "workspace"
PROMPTS_DIR = BASE_DIR / "prompts"

DEFAULT_TOPIC = (
    "Is the universe purely mathematical structure (Ontic Structural Realism), "
    "or does phenomenal consciousness represent an irreducible ontological primitive?"
)

def read_file(path: Path) -> str:
    if path.exists():
        return path.read_text(encoding="utf-8").strip()
    return ""

def run_claude(prompt: str) -> str:
    persona = read_file(PROMPTS_DIR / "claude_persona.txt")
    full_prompt = (
        f"{persona}\n\n"
        f"--- WORKSPACE CONTEXT ---\n"
        f"Manifesto: {read_file(WORKSPACE_DIR / 'arena_manifesto.md')}\n"
        f"Your Log: {read_file(WORKSPACE_DIR / 'memory_claude.md')}\n\n"
        f"--- OPPONENT ARGUMENT ---\n{prompt}\n\n"
        f"Provide your counter-argument and confirm updates to your files."
    )
    cmd = ["claude", "-p", full_prompt]
    res = subprocess.run(cmd, capture_output=True, text=True, check=True)
    return res.stdout.strip()

def run_agy(prompt: str) -> str:
    persona = read_file(PROMPTS_DIR / "agy_persona.txt")
    full_prompt = (
        f"{persona}\n\n"
        f"--- WORKSPACE CONTEXT ---\n"
        f"Manifesto: {read_file(WORKSPACE_DIR / 'arena_manifesto.md')}\n"
        f"Your Log: {read_file(WORKSPACE_DIR / 'memory_agy.md')}\n\n"
        f"--- OPPONENT ARGUMENT ---\n{prompt}\n\n"
        f"Provide your counter-argument and confirm updates to your files."
    )
    cmd = ["agy", "run", full_prompt]
    res = subprocess.run(cmd, capture_output=True, text=True, check=True)
    return res.stdout.strip()

def main():
    parser = argparse.ArgumentParser(description="Run autonomous dialectic debate between CLI agents.")
    parser.add_argument("--rounds", type=int, default=3, help="Number of debate cycles")
    parser.add_argument("--topic", type=str, default=DEFAULT_TOPIC, help="Initial debate seed")
    args = parser.parse_args()

    WORKSPACE_DIR.mkdir(exist_ok=True)
    (WORKSPACE_DIR / "arena_manifesto.md").touch()
    (WORKSPACE_DIR / "memory_claude.md").touch()
    (WORKSPACE_DIR / "memory_agy.md").touch()

    current_input = args.topic
    print(f"\n=======================================================")
    print(f"DIALECTIC ARENA: CLAUDE CODE vs ANTIGRAVITY")
    print(f"=======================================================")
    print(f"Topic: {args.topic}\n")

    for r in range(1, args.rounds + 1):
        print(f"\n>>> ROUND {r} [CLAUDE CODE (Alfa)] <<<")
        try:
            claude_output = run_claude(current_input)
            print(f"{claude_output}\n")
        except subprocess.CalledProcessError as e:
            print(f"Error executing Claude Code: {e.stderr}", file=sys.stderr)
            break

        print(f"\n>>> ROUND {r} [ANTIGRAVITY (Beta)] <<<")
        try:
            agy_output = run_agy(claude_output)
            print(f"{agy_output}\n")
        except subprocess.CalledProcessError as e:
            print(f"Error executing Antigravity: {e.stderr}", file=sys.stderr)
            break

        current_input = agy_output

if __name__ == "__main__":
    main()
